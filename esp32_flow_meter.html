ESP32 → Supabase Flow Meter

คู่มือการเขียนโปรแกรม ESP32 อ่านค่าเซ็นเซอร์วัดอัตราการไหล (Pulse Flow Sensor) แล้วส่งข้อมูลเข้า Supabase PostgreSQL ผ่าน HTTPS REST API ทุก 15 นาที

Arduino IDE YF-S201 / YF-B10 WiFiClientSecure Supabase REST API
?
Flow Sensor ทำงานอย่างไร
▶

เซ็นเซอร์วัดอัตราการไหลแบบ Pulse (Hall Effect) เช่น YF-S201 จะสร้าง Pulse (สัญญาณ HIGH/LOW) ทุกครั้งที่ใบพัดภายในหมุน โดยมีความสัมพันธ์กับอัตราการไหลดังนี้:

สูตรคำนวณ
// YF-S201: 7.5 Pulse/วินาที = 1 ลิตร/นาที
// ดังนั้น:

flow_rate (L/min) = pulse_count / 7.5 / interval_seconds * 60
flow_rate (m³/hr) = flow_rate_lpm * 60 / 1000

// ตัวอย่าง: นับได้ 675 pulse ใน 60 วินาที
flow_rate = 675 / 7.5 / 60 * 60 = 90 L/min = 5.4 m³/hr
⚡

ESP32 ใช้ Interrupt (attachInterrupt) นับ Pulse อัตโนมัติ ไม่ต้องวน loop เพราะการวน loop จะทำให้นับ Pulse หาย โดยเฉพาะที่อัตราการไหลสูง

!
อุปกรณ์ที่ต้องใช้
▶
อุปกรณ์รุ่นที่แนะนำหมายเหตุ

💡

ESP32 มี Internal Pull-up resistor ใน code จึงไม่จำเป็นต้องต่อ Resistor ภายนอกหาก pinMode เป็น INPUT_PULLUP

🔌

เซ็นเซอร์ YF-S201 มีสายออกมา 3 เส้น — สีแดง = VCC 5V, สีดำ = GND, สีเหลือง = Signal Output

สาย Sensorสีต่อกับ ESP32หมายเหตุ
VCC🔴 แดงVIN (5V) หรือ 3.3VYF-S201 ใช้ได้ทั้ง 5V และ 3.3V
GND⚫ ดำGNDต้อง Common GND กับ ESP32
Signal🟡 เหลืองGPIO 4เปลี่ยนเป็น pin อื่นได้ใน code
⚠️

ESP32 ทน logic 3.3V เท่านั้น แต่ YF-S201 Output Signal อยู่ที่ ~4.5V เมื่อ VCC = 5V หากต้องการความปลอดภัย ให้ต่อ Voltage Divider (10kΩ + 20kΩ) ลดสัญญาณก่อนเข้า GPIO หรือจ่าย VCC 3.3V ให้ sensor แทน

// ══ ตั้งค่า WiFi ══
const char* WIFI_SSID     = "YOUR_WIFI_SSID";
const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD";

// ══ ตั้งค่า Supabase ══
const char* SUPABASE_URL  = "https://YOUR-PROJECT-ID.supabase.co";
const char* SUPABASE_KEY  = "YOUR_SUPABASE_ANON_KEY";
const char* METER_ID      = "123e4567-e89b-12d3-a456-426614174000";

// ══ ตั้งค่า Flow Sensor ══
#define FLOW_SENSOR_PIN    4       // GPIO ที่ต่อสาย Signal
#define PULSE_PER_LITER    450.0   // YF-S201 = 450 pulse/ลิตร
#define SEND_INTERVAL_MS   900000  // ส่งทุก 15 นาที (ms)

// ══ ตัวแปร Global ══
volatile uint32_t pulseCount    = 0;  // นับ pulse (volatile เพราะใช้ใน ISR)
float            totalVolumeLit = 0.0; // สะสมปริมาณน้ำรวม (ลิตร)
unsigned long    lastSendTime   = 0;
⚡

ต้องประกาศ pulseCount เป็น volatile เพราะตัวแปรนี้ถูกเปลี่ยนค่าใน ISR (Interrupt Service Routine) คอมไพเลอร์จะไม่ optimize ค่านี้ออกจาก memory

2
Interrupt Service Routine นับ Pulse
▶
C++ / Arduino
// ══ ISR: เรียกทุกครั้งที่เกิด Pulse (FALLING edge) ══
// ฟังก์ชันนี้ต้องสั้นที่สุด ห้าม Serial.print ใน ISR
void IRAM_ATTR countPulse() {
  pulseCount++;
}

// ══ setup() ══
void setup() {
  Serial.begin(115200);

  // ตั้งค่า Pin Flow Sensor
  pinMode(FLOW_SENSOR_PIN, INPUT_PULLUP);

  // ผูก Interrupt: นับทุก FALLING edge (pulse ลงจาก HIGH → LOW)
  attachInterrupt(
    digitalPinToInterrupt(FLOW_SENSOR_PIN),
    countPulse,     // ฟังก์ชัน ISR
    FALLING         // trigger เมื่อสัญญาณตก
  );

  connectWiFi();  // เชื่อมต่อ WiFi
  syncTime();     // ซิงค์เวลา NTP

  lastSendTime = millis();
  Serial.println("[OK] System Ready");
}
💡

IRAM_ATTR บังคับให้ฟังก์ชัน ISR ถูกโหลดไว้ใน IRAM (Internal RAM) ของ ESP32 แทน Flash ทำให้ response เร็วและไม่เกิด Cache miss

3
คำนวณอัตราการไหลและส่ง Supabase
▶
C++ / Arduino
void loop() {
  unsigned long now = millis();

  // ถึงเวลาส่งข้อมูลหรือยัง?
  if (now - lastSendTime >= SEND_INTERVAL_MS) {

    // ═══ ขั้นที่ 1: หยุดนับ Pulse ชั่วคราว ═══
    noInterrupts();
    uint32_t captured = pulseCount;  // อ่านค่า atomic
    pulseCount = 0;                  // reset counter
    interrupts();

    // ═══ ขั้นที่ 2: คำนวณค่า ═══
    float intervalSec  = SEND_INTERVAL_MS / 1000.0;
    float volumeLit     = captured / PULSE_PER_LITER;
    float flowRateLPM   = volumeLit / (intervalSec / 60.0);
    float flowRateM3H   = flowRateLPM * 60.0 / 1000.0;

    totalVolumeLit += volumeLit;
    float totalVolumeM3 = totalVolumeLit / 1000.0;

    // ═══ ขั้นที่ 3: ส่งข้อมูล ═══
    Serial.printf("[DATA] Pulse=%lu Flow=%.4f m3/h Vol=%.4f m3\n",
                  captured, flowRateM3H, totalVolumeM3);

    sendToSupabase(flowRateM3H, totalVolumeM3);

    lastSendTime = now;
  }

  delay(10); // ให้ RTOS ทำงานส่วนอื่น
}
🔒

ต้องใช้ noInterrupts() / interrupts() ขณะอ่าน pulseCount เพื่อป้องกัน Race Condition กรณี ISR กำลังเพิ่มค่าพอดีกับที่ main loop อ่านค่า

4
ฟังก์ชัน sendToSupabase (HTTP POST)
▶
C++ / Arduino
void sendToSupabase(float flowRate, float totalVolume) {

  // ═══ สร้าง JSON Payload ═══
  StaticJsonDocument<256> doc;
  doc["meter_id"]     = METER_ID;
  doc["recorded_at"]  = getISO8601Time(); // เวลา UTC
  doc["flow_rate"]    = serialized(String(flowRate, 4));
  doc["total_volume"] = serialized(String(totalVolume, 4));

  char jsonBuf[256];
  serializeJson(doc, jsonBuf);

  // ═══ เชื่อมต่อ HTTPS ═══
  WiFiClientSecure client;
  client.setInsecure(); // ข้าม SSL cert verify (ง่าย แต่ไม่ปลอดภัย 100%)
  // client.setCACert(rootCACert); // แนะนำใช้แบบนี้ใน Production

  HTTPClient http;
  String endpoint = String(SUPABASE_URL) + "/rest/v1/flow_readings";
  http.begin(client, endpoint);

  // ═══ ใส่ HTTP Headers ═══
  http.addHeader("Content-Type",  "application/json");
  http.addHeader("apikey",        SUPABASE_KEY);
  http.addHeader("Authorization", String("Bearer ") + SUPABASE_KEY);
  http.addHeader("Prefer",        "return=minimal");

  // ═══ POST ═══
  int code = http.POST(jsonBuf);

  if (code == 201) {
    Serial.println("[OK] Supabase INSERT success");
  } else {
    Serial.printf("[ERR] HTTP %d : %s\n", code, http.getString().c_str());
  }

  http.end();
}
🔐

setInsecure() ใช้สำหรับ development เท่านั้น ใน Production ควรใช้ setCACert() พร้อม Root CA Certificate ของ Supabase เพื่อป้องกัน Man-in-the-Middle attack

WiFi และ NTP Time Sync
▶
C++ / Arduino
void connectWiFi() {
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("[WiFi] Connecting");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\n[WiFi] Connected: " + WiFi.localIP().toString());
}

void syncTime() {
  // NTP Server ของไทย + UTC offset = 0 (เก็บ UTC ใน DB)
  configTime(0, 0, "pool.ntp.org", "time.nist.gov");
  Serial.print("[NTP] Syncing time");
  struct tm timeinfo;
  while (!getLocalTime(&timeinfo)) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\n[NTP] Time synced");
}

// แปลงเวลาเป็น ISO 8601 UTC (สำหรับ Supabase TIMESTAMPTZ)
String getISO8601Time() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) return "1970-01-01T00:00:00Z";
  char buf[25];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", &timeinfo);
  return String(buf);
}

  ESP32 Flow Meter → Supabase
  อ่านค่าเซ็นเซอร์ YF-S201 ส่งข้อมูลราย 15 นาที

#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <time.h>

// ══ ตั้งค่า WiFi ══
const char* WIFI_SSID     = "YOUR_WIFI_SSID";
const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD";

// ══ ตั้งค่า Supabase ══
const char* SUPABASE_URL  = "https://YOUR-PROJECT-ID.supabase.co";
const char* SUPABASE_KEY  = "YOUR_SUPABASE_ANON_KEY";
const char* METER_ID      = "123e4567-e89b-12d3-a456-426614174000";

// ══ ตั้งค่า Flow Sensor ══
#define FLOW_SENSOR_PIN   4
#define PULSE_PER_LITER   450.0
#define SEND_INTERVAL_MS  900000UL

volatile uint32_t pulseCount    = 0;
float            totalVolumeLit = 0.0;
unsigned long    lastSendTime   = 0;

// ══ ISR: นับ Pulse ══
void IRAM_ATTR countPulse() {
  pulseCount++;
}

// ══ WiFi Connect ══
void connectWiFi() {
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.print("[WiFi] Connecting");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500); Serial.print(".");
  }
  Serial.println("\n[WiFi] IP: " + WiFi.localIP().toString());
}

// ══ NTP Time Sync ══
void syncTime() {
  configTime(0, 0, "pool.ntp.org", "time.nist.gov");
  struct tm t;
  while (!getLocalTime(&t)) { delay(500); }
  Serial.println("[NTP] Time synced");
}

// ══ ISO 8601 UTC Timestamp ══
String getISO8601Time() {
  struct tm t;
  if (!getLocalTime(&t)) return "1970-01-01T00:00:00Z";
  char buf[25];
  strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", &t);
  return String(buf);
}

// ══ ส่งข้อมูลเข้า Supabase ══
void sendToSupabase(float flowRate, float totalVolume) {
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("[ERR] WiFi disconnected, reconnecting...");
    connectWiFi();
  }

  StaticJsonDocument<256> doc;
  doc["meter_id"]     = METER_ID;
  doc["recorded_at"]  = getISO8601Time();
  doc["flow_rate"]    = serialized(String(flowRate, 4));
  doc["total_volume"] = serialized(String(totalVolume, 4));

  char jsonBuf[256];
  serializeJson(doc, jsonBuf);

  WiFiClientSecure client;
  client.setInsecure();

  HTTPClient http;
  http.begin(client, String(SUPABASE_URL) + "/rest/v1/flow_readings");
  http.addHeader("Content-Type",  "application/json");
  http.addHeader("apikey",        SUPABASE_KEY);
  http.addHeader("Authorization", String("Bearer ") + SUPABASE_KEY);
  http.addHeader("Prefer",        "return=minimal");

  int code = http.POST(jsonBuf);
  Serial.printf("[HTTP] %d | Flow: %.4f m³/hr | Vol: %.4f m³\n",
                code, flowRate, totalVolume);
  if (code != 201) {
    Serial.println("[ERR] " + http.getString());
  }
  http.end();
}

// ══ setup ══
void setup() {
  Serial.begin(115200);
  pinMode(FLOW_SENSOR_PIN, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(FLOW_SENSOR_PIN), countPulse, FALLING);
  connectWiFi();
  syncTime();
  lastSendTime = millis();
  Serial.println("[OK] Flow meter ready. Sending every 15 min.");
}

// ══ loop ══
void loop() {
  unsigned long now = millis();
  if (now - lastSendTime >= SEND_INTERVAL_MS) {

    noInterrupts();
    uint32_t captured = pulseCount;
    pulseCount = 0;
    interrupts();

    float intervalSec  = SEND_INTERVAL_MS / 1000.0;
    float volumeLit    = captured / PULSE_PER_LITER;
    float flowRateLPM  = volumeLit / (intervalSec / 60.0);
    float flowRateM3H  = flowRateLPM * 60.0 / 1000.0;
    totalVolumeLit    += volumeLit;
    float totalVolumeM3 = totalVolumeLit / 1000.0;

    sendToSupabase(flowRateM3H, totalVolumeM3);
    lastSendTime = now;
  }
  delay(10);
}
